Number(input.value) : 30; } function getInputHeight() { const input = document.querySelector("#inputHeight"); return input ? Number(input.value) : 10; } // ============================== // 加速度取得 // ============================== function getAcceleration(distance) { if (!currentData) return 0; const accelerationList = currentData.accelerationsList || []; if (distance < 0) return 0; if (distance >= accelerationList.length) { return 0; } return accelerationList[distance] || 0; } // ============================== // 配線可能判定 // 公式ロジックを簡略化せず使用 // ============================== function isWirable(delay, height, column = 0) { if (!currentData) return 3; const landingKey = document.querySelector("#landingSelect")?.value || Object.keys(landings)[0]; const landing = landings[landingKey]; if (!landing) { console.warn("landing data が見つかりません"); return 3; } let bestError = 3; function checkDelay(totalDelay) { const difference = Math.abs(totalDelay - delay); if (difference === 0) return 0; if (difference <= 1) return 1; if (difference <= 2) return 2; return 3; } function searchDown( baseDelay, baseDistance, dDelay, depth ) { if (depth > 3) return; for (const gap of gaps) { const distance = baseDistance - (gap.down + depth * 2); if (distance < 0) continue; const nextDelay = baseDelay + gap.delay + dDelay + getAcceleration(distance); const error = checkDelay(nextDelay); } } for (const upOption of upOptions) { const distance = height - (upOption.up || 0); if (distance < 0) continue; const startingDelay = upOption.delay || 0; const error = checkDelay(startingDelay); bestError = Math.min(bestError, error); function classify(result) { const difference = Math.abs(result.delay - delay); if (difference === 0) { results.exact.push(result); } else if (difference <= 1) { results.near.push(result); } else if (difference <= 2) { results.far.push(result); const nextDDelay = dDelay + (relativeRailDelays[ (gap.offset + depth - 1) % 3 ] || 0); searchDown( currentDelay, distance, nextDDelay, depth + 1, nextText ); } } for (const upOption of upOptions) { const distance = height - (upOption.up || 0); if (distance < 0) continue; const startingDelay = (upOption.delay || 0); const wiringText = upOption.str || "↑"; classify({ delay: Math.round(startingDelay), height, wiring: wiringText }); searchDown( startingDelay, distance, 0, 0, wiringText ); } return results; } // ============================== // 検索実行 // ============================== function findWiring() { const delay = getInputDelay(); const height = getInputHeight(); const results = findWiringResults(delay, height);
